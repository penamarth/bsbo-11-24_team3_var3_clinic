# Диаграмма классов ИС «ПолиКлик»

![Рис. Классы 1. Реализованное ядро ИС «ПолиКлик»](img/classes-all.png)

Показано ядро из `src/PoliClick`: запись пациента, ведение приема и автоматическая очередь. На диаграмме оставлены основные свойства и методы; параметры части методов сокращены до `...`. Медицинская история пациента состоит из завершенных `Appointment`.

```plantuml
@startuml classes-all
allowmixing
skinparam classAttributeIconSize 0
skinparam linetype polyline
skinparam nodesep 55
hide empty members

rectangle "Поликлиника" as ClinicPart {
  class Clinic {
    + GetSpecialties(): IReadOnlyList<string>
    + GetDoctors(...): IReadOnlyList<Doctor>
    + GetFreeDates(...): IReadOnlyList<DateOnly>
    + CreateSchedule(...): IReadOnlyList<TimeSlot>
    + FindFreeSlots(...): IReadOnlyList<TimeSlot>
    + BookAppointment(...): Appointment
    + OpenAppointment(...): Appointment
    + StartAppointment(...): Appointment
    + CancelAppointment(...): void
    + AddQueueBoard(...): QueueBoard
    + UpdateQueue(): IReadOnlyList<Appointment>
    + MarkNoShow(...): void
  }
}

rectangle "Пациенты" as PatientsPart {
  class PatientStorage {
    ~ AddPatient(...): Patient
    + GetPatient(patientId: int): Patient
  }

  class Patient {
    + Id: int
    + FullName: string
    + BirthDate: DateOnly
    + PolicyNumber: string
    + IsAttached: bool
    --
    + HasActiveAppointment(specialty: string): bool
    + GetMedicalHistory(): IReadOnlyList<Appointment>
  }
}

rectangle "Врачи" as DoctorsPart {
  class DoctorStorage {
    + GetSpecialties(): IReadOnlyList<string>
    + GetDoctors(specialty: string): IReadOnlyList<Doctor>
    ~ AddDoctor(...): Doctor
    + GetDoctor(doctorId: int): Doctor
  }

  class Doctor {
    + Id: int
    + FullName: string
    + Specialty: string
    + SlotMinutes: int
  }
}

rectangle "Расписание и прием" as SchedulePart {
  class SlotStorage {
    + GetFreeDates(...): IReadOnlyList<DateOnly>
    ~ CreateSlots(...): IReadOnlyList<TimeSlot>
    + FindFreeSlots(...): IReadOnlyList<TimeSlot>
  }

  class TimeSlot {
    + Id: int
    + Start: DateTime
    + End: DateTime
    + CabinetNumber: string
    + IsFree: bool
  }

  class Appointment {
    + Id: int
    + Number: string
    + CabinetNumber: string
    + Status: AppointmentStatus
    + CalledAt: DateTime?
    + StartedAt: DateTime?
    + FinishedAt: DateTime?
    + Complaints: string
    + Anamnesis: string
    + Examination: string
    + DiagnosisCode: string
    + Prescriptions: IReadOnlyList<string>
    --
    ~ Call(calledAt: DateTime): void
    ~ MarkNoShow(currentTime: DateTime): void
    + SetExamination(...): void
    + SetDiagnosis(code: string): void
    + AddPrescription(prescription: string): void
    + Complete(): void
  }

  enum AppointmentStatus {
    PLANNED
    IN_PROGRESS
    COMPLETED
    CANCELLED
    NO_SHOW
  }
}

rectangle "Электронная очередь" as QueuePart {
  class QueueBoard {
    + Id: string
    + CabinetNumbers: IReadOnlyList<string>
    + Calls: IReadOnlyDictionary<string, string>
    --
    ~ ShowCall(...): void
    ~ Clear(): void
  }
}

Clinic "1" *-down- "1" PatientStorage
Clinic "1" *-down- "1" SlotStorage
Clinic "1" *-down- "1" DoctorStorage
Clinic "1" *-right- "0..*" QueueBoard
Clinic "1" -down-> "0..*" Appointment : записи
PatientStorage "1" o-down- "0..*" Patient
DoctorStorage "1" o-down- "0..*" Doctor
SlotStorage "1" *-down- "0..*" TimeSlot
Patient "1" -right- "0..*" Appointment : записи пациента
TimeSlot "0..*" -right-> "1" Doctor
Appointment "0..*" -right-> "1" Doctor
TimeSlot "1" -down- "0..*" Appointment : неотмененных\nне больше одной
Appointment .down.> AppointmentStatus

PatientStorage -[hidden]right- SlotStorage
SlotStorage -[hidden]right- DoctorStorage

@enduml
```
