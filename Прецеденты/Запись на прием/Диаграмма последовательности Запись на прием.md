# Диаграмма последовательности «Запись на прием»

Основной успешный сценарий прецедента [«Запись на прием»](Запись%20на%20прием.md), шаги 1–8, и расширение 7а: слот успели занять, пока пациент подтверждал запись. Отмена, перенос и остальные расширения описаны в тексте прецедента.

Пациент работает с экраном записи `:AppointmentUI`. Это портал, мобильное приложение, инфомат или рабочее место медрегистратора: все они вызывают одни и те же системные операции `:AppointmentService`. Слоты хранятся у врача, поэтому сервис берет врачей из пула `:DoctorStorage`, а к слотам обращается только через `:Doctor`. `:Doctor` здесь это данные врача в ПолиКлике вместе с его слотами, а не сам врач: в записи на прием врач не участвует, записавшихся пациентов ПолиКлик сам показывает ему в списке на день.

```plantuml
@startuml booking-sequence
actor Пациент
boundary ":AppointmentUI" as UI
participant ":AppointmentService" as S
participant ":DoctorStorage" as DS
participant ":Doctor" as D
participant ":TimeSlot" as Slot
participant ":Appointment" as A
participant ":PatientStorage" as PS
participant ":Patient" as P
participant ":NotificationService" as N

ref over Пациент, UI, S : Идентификация пациента

Пациент -> UI : начинает запись
UI -> S ++ : getSpecialties(patientId)
S -> DS ++ : getSpecialties()
DS --> S -- : специальности врачей
S --> UI -- : специальности без направления (ПРАВ6)
UI --> Пациент : список специальностей

Пациент -> UI : выбирает специальность\nи способ поиска
UI -> S ++ : findFreeSlots(specialtyId, doctorId, dateFrom, dateTo)
note right of S
  по врачу: doctorId задан
  по дате: dateFrom = dateTo
  ближайшее время: весь горизонт
  в 14 дней (ПРАВ8)
end note
S -> DS ++ : findDoctors(specialtyId, doctorId)
DS --> S -- : врачи специальности
loop для каждого врача
  S -> D ++ : getFreeSlots(dateFrom, dateTo)
  loop для каждого слота в периоде
    D -> Slot ++ : isFree()
    Slot --> D -- : да / нет
  end
  D --> S -- : свободные слоты врача
end
S --> UI -- : свободные слоты
UI --> Пациент : свободные слоты\n(врачи и дни без слотов не показываются)

Пациент -> UI : выбирает слот
UI -> S ++ : selectSlot(patientId, doctorId, slotId)
S -> DS ++ : getDoctor(doctorId)
DS --> S -- : doctor
S -> PS ++ : getPatient(patientId)
PS --> S -- : patient
S -> P ++ : hasActiveAppointment(doctor.specialty)
P --> S -- : нет (ПРАВ7)
S --> UI -- : сведения о записи
UI --> Пациент : сведения о записи\nдля подтверждения

Пациент -> UI : подтверждает запись
UI -> S ++ : bookAppointment(patientId, doctorId, slotId)
S -> DS ++ : getDoctor(doctorId)
DS --> S -- : doctor
S -> PS ++ : getPatient(patientId)
PS --> S -- : patient
S -> D ++ : bookSlot(slotId, patient)
D -> Slot ++ : book(patient)
opt слот еще свободен
  Slot -> A ** : create(patient, slot)
end
Slot --> D -- : appointment или «занято»
D --> S -- : appointment или «занято»
alt слот был свободен
  S -> P ++ : addAppointment(appointment)
  P --> S --
  S -> N ++ : sendConfirmation(appointment)
  N --> S --
  S -> N ++ : scheduleReminder(appointment)
  N --> S --
  S --> UI : appointment
  UI --> Пациент : талон: номер записи, врач,\nкабинет, дата и время
else слот уже занял другой пациент (7а)
  S --> UI : «занято»
  UI --> Пациент : время уже занято,\nобновленный список слотов (шаг 3)
end
deactivate S

@enduml
```

## Системные операции

| Шаг сценария | Операция | Что делает |
| --- | --- | --- |
| 1 | включаемый прецедент «Идентификация пациента» | Пациент входит через ЕСИА, ПолиКлик определяет, кто он |
| 2 | `getSpecialties(patientId)` | Берет специальности врачей из пула врачей и возвращает те, к которым пациент может записаться без направления (ПРАВ6) |
| 3 | `findFreeSlots(specialtyId, doctorId, dateFrom, dateTo)` | Берет из пула врачей нужной специальности и собирает их свободные слоты. Способ поиска (по врачу, по дате, ближайшее время) задается параметрами |
| 4–5 | `selectSlot(patientId, doctorId, slotId)` | Проверяет, что у пациента нет активной записи к специальности этого врача (ПРАВ7), и возвращает сведения для подтверждения |
| 6–8 | `bookAppointment(patientId, doctorId, slotId)` | Находит слот в расписании врача и закрепляет его за пациентом, создает запись со статусом «запланирована», отправляет подтверждение и планирует напоминание о приеме |
